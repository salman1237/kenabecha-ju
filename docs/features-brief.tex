\section{Accounts \& Authentication}
\begin{itemize}
  \item JU-verified signup
  \item OTP email verification
  \item Google one-click sign-in
  \item Profile completion gate
  \item Secure sessions \& password reset
  \item Roles: user, moderator, admin
\end{itemize}

\section{Listings}
\begin{itemize}
  \item Personal and shop-based listings
  \item Rich listing data \& photos
  \item Seller-controlled status (sold, out-of-stock, pause, relist)
  \item Manual shop ordering
  \item Restock requests
  \item Automatic expiry, renewal, featuring \& top picks
  \item Related listings
\end{itemize}

\section{Shops}
\begin{itemize}
  \item Public shop storefronts
  \item Shop dashboard \& branding
  \item Follow a shop
  \item Multiple shops per seller
\end{itemize}

\section{Shop Posts}
\begin{itemize}
  \item Product-anchored posts, re-postable anytime
  \item Rich text (bold/italic/underline, font, colour)
  \item Follow-ranked public feed
  \item Same moderation \& notifications as listings
  \item Edits resubmit for review
\end{itemize}

\section{Search \& Discovery}
\begin{itemize}
  \item Full-text search
  \item Autocomplete suggestions
  \item Filters \& sort
  \item Category taxonomy
  \item View counts \& trending
\end{itemize}

\section{Chat}
\begin{itemize}
  \item Real-time messaging
  \item Per-listing conversations
  \item Owner inbox with shop filters
  \item Offline message email alerts
\end{itemize}

\section{Ratings \& Trust}
\begin{itemize}
  \item Post-sale reviews
  \item Discoverable from shop or listing
  \item Star ratings with review text
\end{itemize}

\section{AI Shopping Assistant}
